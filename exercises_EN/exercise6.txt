# Calculate and evaluate relationships using correlation for a new kind of sweetener.

#----------------------------------------------------------------------------------------------
# CORRELATIONS
#----------------------------------------------------------------------------------------------
# generating the dataset

sugar_amount <- seq(1,13)
optical_rotation <- seq(1, 45.5, by=3.5)
clarity <- seq(9.5,1, by= -0.7)
set.seed(123) # reproducible random data
pH <- rnorm(13, mean=7.5, sd=0.2)
sweetness <- c(2,2,3,4,5,6,6,6,7,9,9,9,8)
pleasantness <- c(5,5,5,7,9,9,9,8,9,7,7,6,5)
sweet_water <- data.frame(sugar_amount,sweetness,optical_rotation,clarity,pleasantness,pH)

# correlation (up to 0.1 trivial, 0.11-0.3 weak, 0.31-0.6 moderate, above 0.61 strong)
cor(sugar_amount,clarity,method="spearman")
cor(sweet_water,method="spearman") # kendall, pearson

# plots of the relationships
plot (sugar_amount,optical_rotation, type="b")
plot (sugar_amount,clarity, type="b")
plot (sugar_amount,pleasantness, type="b")

#----------------------------------------------------------------------------------------------
# SIMPLE LINEAR REGRESSION
#----------------------------------------------------------------------------------------------

# linear regression model (sweetness as a function of the amount of sugar)
# y = b2*x + b1
lm(sweetness~sugar_amount)
plot(sweetness~sugar_amount)
abline(lm(sweetness~sugar_amount))
summary(lm(sweetness~sugar_amount))
