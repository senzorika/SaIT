# categorical data in a matrix
data <- matrix(c(15,2,8,7,14,4,5,5,15,11,5,9), ncol=3, byrow=TRUE) # ncol = numbers per row, byrow=TRUERUE
colnames(data) <- c("A", "B", "C") # column names (products)
rownames(data) <- c("Men", "Women", "Children", "Seniors") # row names (target groups of the product)

# base R cannot compute correspondence analysis, so the MASS package is needed
library(MASS)

# simple correspondence analysis
age <- corresp(data, nf=2) # nf = number of factors

# plot
biplot(age)
