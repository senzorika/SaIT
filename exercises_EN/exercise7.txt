# PCA - principal component analysis
# last modified 11.11.12

# rows = respondents (labels A..H2)
labels <- c("A","B","C","D","E","F","G","H","A2","B2","C2","D2","E2","F2","G2","H2")
sensory <- c(3,7,7,5,5,6,5,6,3,3,2,5,2,7,1,2)
price <- c(5,3,4,7,7,4,7,5,5,3,6,7,4,5,6,3)

brand <- c(7,3,3,3,5,3,3,4,3,5,3,3,6,5,5,7)
data <- data.frame(sensory,price,package_design,brand)
pca <- princomp(data, cor=TRUE)
summary(pca, loadings=TRUE)
pca$scores
biplot(pca, xlabs=labels, cex=0.8, col = c("black", "magenta"), main = "Principal component map", xlab = "Comp.1", ylab = "Comp.2")
