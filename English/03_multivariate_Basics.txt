# ============================================================================
# THEORETICAL BACKGROUND
# ============================================================================
# 
# MULTIVARIATE METHODS analyze multiple variables simultaneously:
# 
# 1. DIMENSION REDUCTION:
#    - PCA: Reduces correlated variables to uncorrelated principal components
#    - CA: Analyzes relationships in contingency tables
#    - MFA: Multiple Factor Analysis for multiple data blocks
# 
# 2. CLUSTERING:
#    - Hierarchical: Creates tree-like structure (dendrogram)
#    - K-means: Partitions data into k clusters
#    - Used for: Product grouping, consumer segmentation
# 
# 3. MULTIVARIATE TESTING:
#    - MANOVA: Multiple dependent variables tested simultaneously
#    - Better than multiple ANOVAs (controls Type I error)
# 
# 4. PREDICTION:
#    - Multiple Regression: Predicts outcome from multiple predictors
#    - Canonical Correlation: Relates two sets of variables
#    - PLS: Partial Least Squares for prediction with many predictors
# 
# WHEN TO USE EACH METHOD:
# - PCA: Understanding structure, reducing dimensions
# - Clustering: Grouping similar products/consumers
# - MANOVA: Testing group differences on multiple attributes
# - Regression: Predicting sensory from instrumental data
# - CA: Analyzing preference data, frequency tables
# 
# ============================================================================

# Example: Descriptive sensory analysis of 8 cheese samples
# 12 trained panelists evaluated on 10 sensory attributes

set.seed(123)

# Create realistic cheese sensory data
cheese_products = paste("Cheese", 1:8)
attributes = c("Yellowness", "Firmness", "Elasticity", "Saltiness", 
               "Sourness", "Creaminess", "Butter_flavor", "Aged_flavor",
               "Smooth_texture", "Persistent_taste")

# Generate correlated sensory data
cheese_pca_data = data.frame(
  Product = cheese_products,
  Yellowness = c(7.2, 8.1, 6.5, 7.8, 5.9, 6.8, 7.5, 8.3),
  Firmness = c(6.8, 7.5, 5.2, 7.1, 4.8, 6.2, 7.0, 7.8),
  Elasticity = c(6.5, 7.2, 5.8, 6.9, 5.5, 6.5, 6.8, 7.5),
  Saltiness = c(5.5, 6.2, 7.8, 6.0, 8.2, 7.5, 6.5, 5.8),
  Sourness = c(4.2, 4.8, 7.5, 5.1, 8.1, 7.2, 5.5, 4.5),
  Creaminess = c(7.8, 8.5, 5.5, 7.2, 4.9, 5.8, 7.5, 8.8),
  Butter_flavor = c(7.5, 8.2, 5.2, 7.0, 4.5, 5.5, 7.2, 8.5),
  Aged_flavor = c(3.5, 4.2, 8.5, 4.8, 9.2, 8.8, 5.2, 4.0),
  Smooth_texture = c(8.2, 8.8, 6.2, 7.8, 5.5, 6.5, 8.0, 9.0),
  Persistent_taste = c(5.8, 6.5, 8.8, 6.2, 9.5, 9.0, 6.8, 6.0)
)

# Display data
cat("\n=== SENSORY DATA ===\n")
print(cheese_pca_data)

# 1. Descriptive statistics
cat("\n=== DESCRIPTIVE STATISTICS ===\n")
summary(cheese_pca_data[,-1])

# 2. Correlation matrix
cat("\n=== CORRELATION MATRIX ===\n")
cor_matrix = cor(cheese_pca_data[,-1])
print(round(cor_matrix, 2))

# Visualize correlations
corrplot(cor_matrix, method="color", type="upper", 
         tl.col="black", tl.srt=45,
         addCoef.col = "black", number.cex=0.7,
         title="Correlation Between Sensory Attributes",
         mar=c(0,0,2,0))

# 3. Perform PCA
cat("\n=== PRINCIPAL COMPONENT ANALYSIS ===\n")

# Standardize data (scale=TRUE) for PCA
pca_result = PCA(cheese_pca_data[,-1], scale.unit=TRUE, graph=FALSE)

# Eigenvalues (variance explained)
cat("\nEigenvalues and Variance Explained:\n")
print(pca_result$eig)

# Scree plot
fviz_eig(pca_result, addlabels=TRUE, ylim=c(0,70),
         main="Scree Plot - Variance Explained by Each PC")

# 4. Component loadings
cat("\n=== VARIABLE LOADINGS (CORRELATIONS WITH PCs) ===\n")
loadings = pca_result$var$coord
print(round(loadings[,1:3], 2))

# 5. Biplot (products and variables)
cat("\n=== BIPLOT ===\n")
fviz_pca_biplot(pca_result, 
                repel=TRUE,
                col.var="blue",
                col.ind="red",
                title="PCA Biplot - Cheese Products and Attributes",
                labelsize=3)

# Variables only
fviz_pca_var(pca_result, 
             col.var="contrib",
             gradient.cols=c("#00AFBB", "#E7B800", "#FC4E07"),
             repel=TRUE,
             title="Variables - PCA")

# Products only with labels
fviz_pca_ind(pca_result,
             col.ind="cos2",
             gradient.cols=c("#00AFBB", "#E7B800", "#FC4E07"),
             repel=TRUE,
             label="all",
             labelsize=4,
             title="Products - PCA")

# 6. Contributions

             title="Contribution to PC1")

# 7. Interpretation
cat("\n=== INTERPRETATION ===\n")

    round(sum(pca_result$eig[1:2,2]), 1), "%\n")

cat("\nPC1 interpretation (", round(pca_result$eig[1,2], 1), "% variance):\n", sep="")

pc1_sorted = sort(abs(pc1_loadings), decreasing=TRUE)
cat("Main positive contributors:", 
    paste(names(pc1_sorted)[pc1_loadings[names(pc1_sorted)][1:3] > 0], 
          collapse=", "), "\n")
cat("Main negative contributors:", 
    paste(names(pc1_sorted)[pc1_loadings[names(pc1_sorted)][1:3] < 0], 
          collapse=", "), "\n")

# Using the same cheese data for clustering

# 1. Calculate distance matrix
cat("\n=== DISTANCE MATRIX ===\n")
# Euclidean distance on scaled data
scaled_data = scale(cheese_pca_data[,-1])
dist_matrix = dist(scaled_data, method="euclidean")
print(round(as.matrix(dist_matrix), 2))

# 2. Perform hierarchical clustering
cat("\n=== HIERARCHICAL CLUSTERING ===\n")

# Different linkage methods
hc_complete = hclust(dist_matrix, method="complete")
hc_average = hclust(dist_matrix, method="average")
hc_ward = hclust(dist_matrix, method="ward.D2")

# Plot dendrograms
par(mfrow=c(2,2))
plot(hc_complete, labels=cheese_products, main="Complete Linkage",
     xlab="", sub="", cex=0.9)
rect.hclust(hc_complete, k=3, border="red")

plot(hc_average, labels=cheese_products, main="Average Linkage",
     xlab="", sub="", cex=0.9)
rect.hclust(hc_average, k=3, border="blue")

plot(hc_ward, labels=cheese_products, main="Ward's Method",
     xlab="", sub="", cex=0.9)
rect.hclust(hc_ward, k=3, border="green")

# Cophenetic correlation (goodness of fit)
coph_complete = cor(dist_matrix, cophenetic(hc_complete))
coph_average = cor(dist_matrix, cophenetic(hc_average))
coph_ward = cor(dist_matrix, cophenetic(hc_ward))

plot(1, type="n", xlim=c(0,1), ylim=c(0,1), 
     xlab="", ylab="", main="Cophenetic Correlation",
     axes=FALSE)
text(0.5, 0.7, paste("Complete:", round(coph_complete, 3)))
text(0.5, 0.5, paste("Average:", round(coph_average, 3)))
text(0.5, 0.3, paste("Ward:", round(coph_ward, 3)))
text(0.5, 0.1, "Higher = Better fit", col="blue")

par(mfrow=c(1,1))

cat("\nCophenetic correlations (higher = better):\n")
cat("Complete linkage:", round(coph_complete, 3), "\n")
cat("Average linkage:", round(coph_average, 3), "\n")
cat("Ward's method:", round(coph_ward, 3), "\n")

# 3. Cut tree to create groups
cat("\n=== CLUSTER ASSIGNMENT ===\n")
clusters_3 = cutree(hc_ward, k=3)
cluster_df = data.frame(Product=cheese_products, Cluster=clusters_3)
print(cluster_df)

# 4. Enhanced dendrogram
library(dendextend)
dend = as.dendrogram(hc_ward)
dend_colored = color_branches(dend, k=3)
plot(dend_colored, main="Ward's Hierarchical Clustering (3 Clusters)")

# 5. Cluster profiling
cat("\n=== CLUSTER PROFILES ===\n")
cheese_with_clusters = cbind(cheese_pca_data, Cluster=as.factor(clusters_3))
cluster_profiles = aggregate(.~Cluster, data=cheese_with_clusters[,-1], mean)
print(round(cluster_profiles, 2))

# Visualize cluster profiles
cluster_long = reshape2::melt(cluster_profiles, id.vars="Cluster")
ggplot(cluster_long, aes(x=variable, y=value, color=Cluster, group=Cluster)) +
  geom_line(size=1) +
  geom_point(size=3) +
  theme_minimal() +
  theme(axis.text.x = element_text(angle=45, hjust=1)) +
  labs(title="Cluster Profiles - Mean Attribute Scores",
       x="Attribute", y="Mean Score") +
  scale_color_manual(values=c("red", "blue", "green"))

# Example: 3 wine types evaluated on 4 attributes by different panels

set.seed(456)
n_per_group = 20

wine_data = data.frame(
  Wine_Type = factor(rep(c("Red", "White", "Rosé"), each=n_per_group)),
  Sweetness = c(rnorm(n_per_group, 3.5, 0.8),   # Red
                rnorm(n_per_group, 5.2, 0.9),   # White
                rnorm(n_per_group, 6.5, 0.7)),  # Rosé
  Acidity = c(rnorm(n_per_group, 7.2, 0.9),     # Red
              rnorm(n_per_group, 7.8, 0.8),     # White
              rnorm(n_per_group, 5.5, 0.7)),    # Rosé
  Body = c(rnorm(n_per_group, 7.5, 0.8),        # Red
           rnorm(n_per_group, 4.2, 0.9),        # White
           rnorm(n_per_group, 5.5, 0.8)),       # Rosé
  Astringency = c(rnorm(n_per_group, 6.8, 1.0), # Red
                  rnorm(n_per_group, 3.2, 0.8), # White
                  rnorm(n_per_group, 4.5, 0.9)) # Rosé
)

# Descriptive statistics
cat("\n=== DESCRIPTIVE STATISTICS BY WINE TYPE ===\n")
aggregate(.~Wine_Type, data=wine_data, mean)

# Visualize
wine_long = reshape2::melt(wine_data, id.vars="Wine_Type")
ggplot(wine_long, aes(x=Wine_Type, y=value, fill=Wine_Type)) +
  geom_boxplot() +
  facet_wrap(~variable, scales="free_y") +
  theme_minimal() +
  labs(title="Sensory Attributes by Wine Type",
       y="Score") +
  scale_fill_manual(values=c("darkred", "gold", "pink"))

# Check assumptions
cat("\n=== CHECKING ASSUMPTIONS ===\n")

# Multivariate normality (Shapiro-Wilk for each variable by group)
cat("\nNormality tests by group and variable:\n")
for(wine in levels(wine_data$Wine_Type)) {
  cat("\n", wine, ":\n", sep="")
  subset_data = wine_data[wine_data$Wine_Type==wine, -1]
  for(var in names(subset_data)) {
    p_val = shapiro.test(subset_data[[var]])$p.value
    cat("  ", var, ": p =", round(p_val, 4), "\n")
  }
}

# Homogeneity of covariance matrices (Box's M test)
cat("\n=== BOX'S M TEST (Homogeneity of Covariance Matrices) ===\n")
if(require(biotools)) {
  boxm_result = boxM(wine_data[,-1], wine_data$Wine_Type)
  print(boxm_result)
} else {
  cat("Note: Install 'biotools' package for Box's M test\n")
}

# Perform MANOVA
cat("\n=== MANOVA RESULTS ===\n")
manova_result = manova(cbind(Sweetness, Acidity, Body, Astringency) ~ Wine_Type,
                       data=wine_data)
manova_summary = summary(manova_result, test="Wilks")
print(manova_summary)

# Alternative tests
cat("\n=== ALTERNATIVE MANOVA TESTS ===\n")
cat("\nPillai's Trace:\n")
print(summary(manova_result, test="Pillai"))

cat("\nHotelling-Lawley Trace:\n")
print(summary(manova_result, test="Hotelling-Lawley"))

cat("\nRoy's Largest Root:\n")
print(summary(manova_result, test="Roy"))

# Univariate ANOVAs (follow-up)
cat("\n=== UNIVARIATE ANOVA FOLLOW-UPS ===\n")
print(summary.aov(manova_result))

# Effect size
cat("\n=== EFFECT SIZES (Partial Eta-Squared) ===\n")
# Calculate for each variable
for(var in c("Sweetness", "Acidity", "Body", "Astringency")) {
  aov_model = aov(as.formula(paste(var, "~ Wine_Type")), data=wine_data)
  ss = summary(aov_model)[[1]]$"Sum Sq"
  eta_sq = ss[1] / sum(ss)
  cat(var, ":", round(eta_sq, 3), "\n")
}

# Post-hoc tests if significant
if(manova_summary$stats[1,6] < 0.05) {
  cat("\n=== POST-HOC TESTS (Tukey HSD) ===\n")
  for(var in c("Sweetness", "Acidity", "Body", "Astringency")) {
    cat("\n", var, ":\n", sep="")
    aov_model = aov(as.formula(paste(var, "~ Wine_Type")), data=wine_data)
    tukey = TukeyHSD(aov_model)
    print(tukey)
  }
}

# Example: Predicting overall quality from sensory attributes

set.seed(789)
n = 50

regression_data = data.frame(
  Appearance = rnorm(n, 7, 1),
  Aroma = rnorm(n, 6.5, 1.2),
  Flavor = rnorm(n, 7.2, 1.1),
  Texture = rnorm(n, 6.8, 1),
  Aftertaste = rnorm(n, 6.5, 1.3)
)

# Create overall quality as weighted combination + noise
regression_data$Overall_Quality = 0.5 + 
  0.15 * regression_data$Appearance +
  0.25 * regression_data$Aroma +
  0.35 * regression_data$Flavor +
  0.15 * regression_data$Texture +
  0.10 * regression_data$Aftertaste +
  rnorm(n, 0, 0.5)

# Descriptive statistics
cat("\n=== DESCRIPTIVE STATISTICS ===\n")
summary(regression_data)

# Correlation matrix
cat("\n=== CORRELATION MATRIX ===\n")
cor_reg = cor(regression_data)
print(round(cor_reg, 2))

corrplot(cor_reg, method="color", type="upper",
         addCoef.col="black", number.cex=0.8,
         tl.col="black", tl.srt=45,
         title="Correlation Matrix - Quality Attributes",
         mar=c(0,0,2,0))

# Fit multiple regression model
cat("\n=== MULTIPLE REGRESSION MODEL ===\n")
lm_model = lm(Overall_Quality ~ Appearance + Aroma + Flavor + 
                Texture + Aftertaste, data=regression_data)
summary(lm_model)

# Standardized coefficients
cat("\n=== STANDARDIZED COEFFICIENTS ===\n")
lm_scaled = lm(scale(Overall_Quality) ~ scale(Appearance) + scale(Aroma) + 
                 scale(Flavor) + scale(Texture) + scale(Aftertaste), 
               data=regression_data)
print(summary(lm_scaled)$coefficients)

# Model diagnostics
cat("\n=== MODEL DIAGNOSTICS ===\n")
par(mfrow=c(2,2))
plot(lm_model)
par(mfrow=c(1,1))

# Check multicollinearity (VIF)
cat("\n=== MULTICOLLINEARITY CHECK (VIF) ===\n")
vif_values = vif(lm_model)
print(vif_values)
cat("\nInterpretation: VIF > 10 indicates problematic multicollinearity\n")

# Predictions
cat("\n=== MODEL PREDICTIONS ===\n")
predictions = predict(lm_model, regression_data)
comparison = data.frame(
  Observed = regression_data$Overall_Quality,
  Predicted = predictions,
  Residual = residuals(lm_model)
)
head(comparison, 10)

# Plot observed vs predicted
plot(regression_data$Overall_Quality, predictions,
     xlab="Observed Overall Quality", 
     ylab="Predicted Overall Quality",
     main="Observed vs Predicted Quality",
     pch=19, col="blue")
abline(0, 1, col="red", lwd=2)
legend("topleft", legend=c("Perfect fit"), col="red", lwd=2)

# Example: Relating instrumental measurements to sensory attributes

set.seed(321)
n_samples = 40

# Instrumental measurements (Set 1)
instrumental = data.frame(
  Hardness_inst = rnorm(n_samples, 50, 10),
  Cohesiveness_inst = rnorm(n_samples, 0.5, 0.1),
  Springiness_inst = rnorm(n_samples, 0.8, 0.15)
)

# Sensory attributes (Set 2) - correlated with instrumental
sensory = data.frame(
  Firmness_sens = 0.7*instrumental$Hardness_inst + rnorm(n_samples, 20, 5),
  Chewiness_sens = 0.6*instrumental$Cohesiveness_inst*100 + rnorm(n_samples, 30, 8),
  Elasticity_sens = 0.8*instrumental$Springiness_inst*10 + rnorm(n_samples, 5, 2)
)

# Standardize variables
instrumental_scaled = scale(instrumental)
sensory_scaled = scale(sensory)

# Perform CCA
cat("\n=== CANONICAL CORRELATION ANALYSIS ===\n")
if(require(CCA)) {
  cca_result = cc(instrumental_scaled, sensory_scaled)
  
  cat("\nCanonical correlations:\n")
  print(cca_result$cor)
  
  cat("\nCanonical loadings (correlations with canonical variates):\n")
  cat("\nInstrumental variables:\n")
  print(round(cca_result$xcoef, 3))
  
  cat("\nSensory variables:\n")
  print(round(cca_result$ycoef, 3))
  
  # Test significance
  cat("\n=== SIGNIFICANCE TESTS ===\n")
  n = nrow(instrumental)
  p = ncol(instrumental)
  q = ncol(sensory)
  
  for(i in 1:min(p,q)) {
    wilks = prod(1 - cca_result$cor[i:min(p,q)]^2)
    df1 = (p - i + 1) * (q - i + 1)
    df2 = n - 1 - (p + q + 1)/2
    F_stat = ((1 - wilks^(1/2)) / wilks^(1/2)) * (df2/df1)
    p_value = 1 - pf(F_stat, df1, df2)
    
    cat("Canonical dimension", i, ": Wilks' Lambda =", round(wilks, 4),
        ", F =", round(F_stat, 3), ", p =", round(p_value, 4), "\n")
  }
  
  # Visualize
  par(mfrow=c(1,2))
  
  # Canonical variates
  U = instrumental_scaled %*% cca_result$xcoef[,1]
  V = sensory_scaled %*% cca_result$ycoef[,1]
  plot(U, V, 
       xlab="Instrumental Canonical Variate 1",
       ylab="Sensory Canonical Variate 1",
       main=paste("Canonical Correlation =", round(cca_result$cor[1], 3)),
       pch=19, col="blue")
  abline(lm(V~U), col="red", lwd=2)
  
  # Loadings plot
  plot(cca_result$scores$corr.X.xscores[,1], 
       cca_result$scores$corr.Y.xscores[,1],
       xlim=c(-1,1), ylim=c(-1,1),
       xlab="Correlation with Instrumental CV1",
       ylab="Correlation with Sensory CV1",
       main="Variable Loadings",
       type="n")
  text(cca_result$scores$corr.X.xscores[,1], 
       cca_result$scores$corr.Y.xscores[,1],
       labels=c(colnames(instrumental), colnames(sensory)),
       col=c(rep("blue",3), rep("red",3)))
  abline(h=0, v=0, lty=2)
  legend("topleft", legend=c("Instrumental", "Sensory"),
         col=c("blue", "red"), pch=1)
  
  par(mfrow=c(1,1))
}

  Color = c(8, 8.5, 2, 2.5, 5, 5.5, 9, 8.8, 3, 2.8, 6, 5.8),
  Bitterness = c(6, 6.5, 3, 3.5, 5, 5.5, 7, 7.5, 4, 3.8, 5.5, 5.2),
  Alcohol = c(7, 7.5, 4, 4.5, 5.5, 6, 8, 8.5, 4.5, 4.2, 6.5, 6.2),

  Fruitiness = c(3, 3.5, 6, 6.5, 4.5, 4.8, 2.5, 2.8, 6.5, 6.2, 4, 4.2)
)

# a) Standardize
beer_scaled = scale(beer_cluster[,-1])
rownames(beer_scaled) = beer_cluster$Beer

# b) Distance matrix
beer_dist = dist(beer_scaled, method="euclidean")
cat("\n=== b) EUCLIDEAN DISTANCE MATRIX ===\n")
print(round(as.matrix(beer_dist), 2))

# c) Different linkage methods
hc_complete = hclust(beer_dist, method="complete")
hc_average = hclust(beer_dist, method="average")
hc_ward = hclust(beer_dist, method="ward.D2")

par(mfrow=c(2,2))
plot(hc_complete, labels=beer_cluster$Beer, main="Complete Linkage", 
     xlab="", sub="")
rect.hclust(hc_complete, k=3, border="red")

plot(hc_average, labels=beer_cluster$Beer, main="Average Linkage",
     xlab="", sub="")
rect.hclust(hc_average, k=3, border="blue")

plot(hc_ward, labels=beer_cluster$Beer, main="Ward's Method",
     xlab="", sub="")
rect.hclust(hc_ward, k=3, border="green")
par(mfrow=c(1,1))

# d) Cophenetic correlations
coph_comp = cor(beer_dist, cophenetic(hc_complete))
coph_avg = cor(beer_dist, cophenetic(hc_average))
coph_ward = cor(beer_dist, cophenetic(hc_ward))

# e) Cut into 3 clusters
clusters = cutree(hc_ward, k=3)

+-------------------------+---------------------------+----------------------------------+
| GOAL                    | METHOD                    | WHEN TO USE                      |
+-------------------------+---------------------------+----------------------------------+
| Reduce dimensions       | PCA                       | Many correlated variables        |
| Reduce dimensions       | Correspondence Analysis   | Contingency tables               |
| Group products          | Hierarchical Clustering   | Unknown number of groups         |
| Group products          | K-means Clustering        | Known/estimated number of groups |
| Test group differences  | MANOVA                    | Multiple dependent variables     |
| Predict outcome         | Multiple Regression       | One outcome, many predictors     |
| Relate two sets         | Canonical Correlation     | Two sets of variables            |
| Predict with many vars  | PLS Regression            | More predictors than samples     |
+-------------------------+---------------------------+----------------------------------+

KEY DECISION POINTS:

1. What is your goal?
   - Understand structure → PCA
   - Find groups → Clustering
   - Test differences → MANOVA
   - Predict → Regression

2. What type of data?
   - Continuous quantitative → PCA, Clustering, MANOVA, Regression
   - Categorical/frequency → Correspondence Analysis
   - Mixed → Consider separate analyses

3. Data structure?
   - Many correlated variables → PCA first
   - Independent variables → Proceed with analysis
   - Two distinct sets → Canonical Correlation

4. Sample size?
   - Large (n > 100) → All methods OK
   - Medium (30-100) → Use with caution
   - Small (< 30) → Avoid complex methods

BEST PRACTICES:

1. ALWAYS standardize data for:
   - PCA (if variables have different units)
   - Clustering (essential!)
   - Canonical Correlation

2. Check assumptions:
   - Normality (Shapiro-Wilk)
   - Homogeneity (Levene's, Box's M)
   - Linearity (scatterplots)

3. Validate results:
   - Cross-validation
   - Bootstrap
   - Independent dataset

4. Report effect sizes:
   - PCA: % variance explained
   - Clustering: Silhouette scores
   - MANOVA: Partial η²
   - Regression: R²

5. Visualize results:
   - Biplots for PCA
   - Dendrograms for clustering
   - Diagnostic plots for regression
