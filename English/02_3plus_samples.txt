# ============================================================================
# THEORETICAL BACKGROUND
# ============================================================================
# 
# When comparing THREE OR MORE samples:
# 
# A. INDEPENDENT SAMPLES (different panelists for each sample)
#    - Parametric (assumes normality): One-Way ANOVA
#    - Non-parametric: Kruskal-Wallis Test
# 
# B. DEPENDENT SAMPLES (same panelists evaluate all samples)
#    - Parametric (assumes normality): Repeated Measures ANOVA
#    - Non-parametric: Friedman Test
# 
# C. POST-HOC TESTS (after finding significant difference)
#    - Parametric: Tukey HSD, Bonferroni, Scheffé
#    - Non-parametric: Dunn's test, Nemenyi test
# 
# D. DISCRIMINATION TESTS
#    - Triangle test: p = 1/3 (33.3%)
#    - Tetrad test: p = 1/3 (33.3%)
#    - Use Binomial test
# 
# E. CATEGORICAL DATA
#    - Chi-square test for independence
#    - Chi-square goodness of fit
# 
# ============================================================================

# ============================================================================
# PART 1: ONE-WAY ANOVA (INDEPENDENT SAMPLES - PARAMETRIC)
# ============================================================================

cat("\n")
cat("========================================\n")
cat("PART 1: ONE-WAY ANOVA\n")
cat("========================================\n")

# Example: Four different cheese brands evaluated by different consumer panels
# Each brand evaluated by 15 different consumers

set.seed(123)
brand_A = rnorm(15, mean=6.8, sd=1.2)
brand_B = rnorm(15, mean=7.5, sd=1.1)
brand_C = rnorm(15, mean=6.5, sd=1.3)
brand_D = rnorm(15, mean=7.8, sd=1.0)

# Create data frame in long format
cheese_scores = c(brand_A, brand_B, brand_C, brand_D)
cheese_brands = factor(rep(c("Brand A", "Brand B", "Brand C", "Brand D"), each=15))
cheese_data = data.frame(Score = cheese_scores, Brand = cheese_brands)

# 1. Descriptive statistics
cat("\n=== DESCRIPTIVE STATISTICS ===\n")
aggregate(Score ~ Brand, data=cheese_data, FUN=function(x) {
  c(Mean=round(mean(x),2), SD=round(sd(x),2), 
    Median=round(median(x),2), N=length(x))
})

# 2. Visualize data
boxplot(Score ~ Brand, data=cheese_data,
        main="Cheese Brand Comparison",
        ylab="Hedonic Score",
        col=c("lightyellow", "lightblue", "lightgreen", "lightpink"))

# Add means
means = tapply(cheese_data$Score, cheese_data$Brand, mean)
points(1:4, means, col="red", pch=19, cex=1.5)
legend("topleft", legend="Mean", col="red", pch=19)

# 3. Check assumptions
cat("\n=== CHECKING ASSUMPTIONS ===\n")

# Normality for each group
cat("Normality tests (Shapiro-Wilk):\n")
by(cheese_data$Score, cheese_data$Brand, shapiro.test)

# Homogeneity of variance (Levene's test)
cat("\nLevene's test for homogeneity of variances:\n")
levene_result = leveneTest(Score ~ Brand, data=cheese_data)
print(levene_result)

# Visual check - Q-Q plots
par(mfrow=c(2,2))
for(brand in levels(cheese_data$Brand)) {
  qqnorm(cheese_data$Score[cheese_data$Brand==brand], main=paste("Q-Q Plot:", brand))
  qqline(cheese_data$Score[cheese_data$Brand==brand], col="red")
}
par(mfrow=c(1,1))

# 4. One-Way ANOVA
cat("\n=== ONE-WAY ANOVA ===\n")
anova_model = aov(Score ~ Brand, data=cheese_data)
anova_summary = summary(anova_model)
print(anova_summary)

# ANOVA table interpretation
cat("\nInterpretation:\n")
p_value = anova_summary[[1]]$"Pr(>F)"[1]
if(p_value < 0.05) {
  cat("p-value =", round(p_value, 4), "< 0.05\n")
  cat("Conclusion: At least one brand is significantly different from others.\n")
  cat("Proceed to post-hoc tests to identify which brands differ.\n")
} else {
  cat("p-value =", round(p_value, 4), "> 0.05\n")
  cat("Conclusion: No significant difference between brands.\n")
}

# Effect size (Eta-squared)
ss_between = anova_summary[[1]]$"Sum Sq"[1]
ss_total = sum(anova_summary[[1]]$"Sum Sq")
eta_squared = ss_between / ss_total
cat("\nEffect size (η²):", round(eta_squared, 3), "\n")
cat("Interpretation:", 
    ifelse(eta_squared > 0.14, "Large effect",
    ifelse(eta_squared > 0.06, "Medium effect", "Small effect")), "\n")

# Example: Ordinal data or non-normal distributions
# Texture preferences rated on 5-point scale by different consumers

texture_scores = c(3,4,3,4,5,4,3,4,4,3,  # Product A
                   4,5,4,5,5,4,5,4,5,4,  # Product B
                   2,3,2,3,3,2,3,2,3,2,  # Product C
                   5,5,4,5,5,5,4,5,5,4)  # Product D

texture_products = factor(rep(c("Product A", "Product B", 
                                "Product C", "Product D"), each=10))
texture_data = data.frame(Score = texture_scores, Product = texture_products)

# Descriptive statistics
cat("\n=== DESCRIPTIVE STATISTICS ===\n")
aggregate(Score ~ Product, data=texture_data, FUN=function(x) {
  c(Median=median(x), Mean=round(mean(x),2), 
    Q1=quantile(x,0.25), Q3=quantile(x,0.75))
})

# Boxplot
boxplot(Score ~ Product, data=texture_data,
        main="Texture Preference Comparison",
        ylab="Preference Score (1-5)",
        col=rainbow(4))

# Kruskal-Wallis test
cat("\n=== KRUSKAL-WALLIS TEST ===\n")
kw_result = kruskal.test(Score ~ Product, data=texture_data)
print(kw_result)

cat("\nInterpretation:\n")
if(kw_result$p.value < 0.05) {
  cat("p-value =", round(kw_result$p.value, 4), "< 0.05\n")
  cat("Conclusion: At least one product differs significantly in texture preference.\n")
  cat("Proceed to Dunn's test for pairwise comparisons.\n")
} else {
  cat("p-value =", round(kw_result$p.value, 4), "> 0.05\n")
  cat("Conclusion: No significant difference in texture preference.\n")
}

# Example: Same 12 trained panelists evaluate 4 wine samples
# Each panelist evaluates all 4 wines

panelist_id = rep(1:12, times=4)
wine_type = factor(rep(c("Cabernet", "Merlot", "Pinot Noir", "Syrah"), each=12))

# Generate realistic wine scores

wine_effects = rep(c(0, 0.5, -0.3, 0.8), each=12)
wine_scores = base_scores + wine_effects + rnorm(48, 0, 0.5)

wine_data = data.frame(
  Panelist = factor(panelist_id),
  Wine = wine_type,
  Score = wine_scores
)

# Descriptive statistics
cat("\n=== DESCRIPTIVE STATISTICS ===\n")
aggregate(Score ~ Wine, data=wine_data, FUN=function(x) {
  c(Mean=round(mean(x),2), SD=round(sd(x),2))
})

# Visualize
boxplot(Score ~ Wine, data=wine_data,
        main="Wine Preference (Repeated Measures)",
        ylab="Quality Score",
        col=c("darkred", "red", "pink", "purple"))

library(ggplot2)
if(require(ggplot2)) {
  print(ggplot(wine_data, aes(x=Wine, y=Score, group=Panelist, color=Panelist)) +
    geom_line() +
    geom_point() +
    theme_minimal() +
    labs(title="Individual Panelist Profiles"))
}

# Check assumptions
cat("\n=== CHECKING ASSUMPTIONS ===\n")

# Normality of residuals
wine_aov = aov(Score ~ Wine + Error(Panelist/Wine), data=wine_data)

# Sphericity (Mauchly's test) - requires different approach
# Using car package for more detailed analysis
wine_wide = reshape(wine_data, idvar="Panelist", timevar="Wine", direction="wide")
names(wine_wide) = c("Panelist", "Cabernet", "Merlot", "PinotNoir", "Syrah")
wine_matrix = as.matrix(wine_wide[,-1])

# Simple repeated measures ANOVA
cat("\n=== REPEATED MEASURES ANOVA ===\n")
wine_rm = aov(Score ~ Wine + Error(Panelist), data=wine_data)
summary(wine_rm)

# Alternative using lm for more details
wine_lm = lm(Score ~ Wine + Panelist, data=wine_data)
cat("\nANOVA Table:\n")
print(anova(wine_lm))

# PART 4: FRIEDMAN TEST (DEPENDENT SAMPLES - NON-PARAMETRIC)
# ============================================================================

cat("\n\n")
cat("========================================\n")
cat("PART 4: FRIEDMAN TEST\n")
cat("========================================\n")

# Example: Same panelists rate 5 chocolate samples on ordinal scale
# 10 panelists, 5 chocolates

chocolate_data_wide = data.frame(
  Panelist = 1:10,
  Choc_A = c(7, 6, 7, 8, 7, 6, 7, 8, 7, 6),
  Choc_B = c(8, 7, 8, 9, 8, 7, 8, 9, 8, 7),
  Choc_C = c(6, 5, 6, 7, 6, 5, 6, 7, 6, 5),
  Choc_D = c(7, 6, 7, 8, 7, 6, 7, 8, 7, 6),
  Choc_E = c(9, 8, 9, 9, 9, 8, 9, 9, 9, 8)
)

# Convert to long format
chocolate_data_long = reshape(chocolate_data_wide, 
                             varying = list(2:6),
                             v.names = "Score",
                             timevar = "Chocolate",
                             times = c("A", "B", "C", "D", "E"),
                             direction = "long")
chocolate_data_long$Chocolate = factor(chocolate_data_long$Chocolate)

# Descriptive statistics
cat("\n=== DESCRIPTIVE STATISTICS ===\n")
aggregate(Score ~ Chocolate, data=chocolate_data_long, FUN=function(x) {
  c(Median=median(x), Mean=round(mean(x),2), IQR=IQR(x))
})

# Visualize
boxplot(Score ~ Chocolate, data=chocolate_data_long,
        main="Chocolate Preference (Friedman Test)",
        ylab="Preference Score",
        col=c("brown", "chocolate4", "chocolate3", "chocolate2", "chocolate1"))

# Friedman test
cat("\n=== FRIEDMAN TEST ===\n")
friedman_result = friedman.test(Score ~ Chocolate | Panelist, 
                                data=chocolate_data_long)
print(friedman_result)

cat("\nInterpretation:\n")
if(friedman_result$p.value < 0.05) {
  cat("p-value =", round(friedman_result$p.value, 4), "< 0.05\n")
  cat("Conclusion: Significant differences exist among chocolates.\n")
  cat("Proceed to post-hoc Nemenyi test.\n")
} else {
  cat("p-value =", round(friedman_result$p.value, 4), "> 0.05\n")
  cat("Conclusion: No significant differences among chocolates.\n")
}

# Calculate Kendall's W (effect size for Friedman)
k = length(unique(chocolate_data_long$Chocolate))  # number of treatments
n = length(unique(chocolate_data_long$Panelist))   # number of blocks
kendalls_w = friedman_result$statistic / (n * (k - 1))
cat("\nKendall's W (effect size):", round(kendalls_w, 3), "\n")
cat("Interpretation:", 
    ifelse(kendalls_w > 0.5, "Strong agreement",
    ifelse(kendalls_w > 0.3, "Moderate agreement", "Weak agreement")), "\n")

# Example 1: Preference test with 4 ice cream flavors
# 120 consumers chose their favorite

ice_cream_preferences = c(
  Vanilla = 25,
  Chocolate = 45,
  Strawberry = 30,
  Mint = 20
)

cat("\n=== CHI-SQUARE GOODNESS OF FIT TEST ===\n")
cat("Observed frequencies:\n")
print(ice_cream_preferences)

# Test if preferences are equal (uniform distribution)
chisq_uniform = chisq.test(ice_cream_preferences)
print(chisq_uniform)

# Visualize
barplot(ice_cream_preferences, 
        main="Ice Cream Preferences",
        ylab="Number of Consumers",
        col=c("lightyellow", "chocolate4", "pink", "lightgreen"))
abline(h=30, col="red", lty=2, lwd=2)
text(2.5, 32, "Expected (equal preference)", col="red")

cat("\nInterpretation:\n")
if(chisq_uniform$p.value < 0.05) {
  cat("p-value =", round(chisq_uniform$p.value, 4), "< 0.05\n")
  cat("Conclusion: Preferences are NOT equally distributed.\n")
  cat("Chocolate is significantly more preferred.\n")
} else {
  cat("p-value =", round(chisq_uniform$p.value, 4), "> 0.05\n")
  cat("Conclusion: No significant preference differences.\n")
}

# Example 2: Chi-square test of independence
# Gender vs. Coffee preference

cat("\n\n=== CHI-SQUARE TEST OF INDEPENDENCE ===\n")
coffee_gender = matrix(c(30, 20, 15,   # Males
                        25, 35, 25),   # Females
                      nrow=2, byrow=TRUE,
                      dimnames=list(Gender = c("Male", "Female"),
                                   Coffee = c("Espresso", "Cappuccino", "Latte")))

print(coffee_gender)

chisq_independence = chisq.test(coffee_gender)
print(chisq_independence)

# Visualize
barplot(coffee_gender, beside=TRUE, 
        main="Coffee Preference by Gender",
        ylab="Count",
        col=c("lightblue", "lightpink"),
        legend=TRUE)


# ============================================================================
# PART 7: TRIANGLE TEST (DISCRIMINATION WITH 3+ PRESENTATIONS)
# ============================================================================

cat("\n\n")
cat("========================================\n")
cat("PART 7: TRIANGLE TEST\n")
cat("========================================\n")

# Example: Triangle test with 80 panelists
# 35 correctly identified the different sample

cat("\n=== TRIANGLE TEST (p = 1/3) ===\n")
triangle_result = binom.test(35, 80, p=1/3)
print(triangle_result)

cat("\nInterpretation:\n")
if(triangle_result$p.value < 0.05) {
  cat("p-value =", round(triangle_result$p.value, 4), "< 0.05\n")
  cat("Conclusion: Panelists can detect a difference between samples.\n")
} else {
  cat("p-value =", round(triangle_result$p.value, 4), "> 0.05\n")
  cat("Conclusion: No detectable difference between samples.\n")
}

# Calculate minimum needed for significance
cat("\nMinimum correct responses needed for significance:\n")
for(i in 30:40) {
  p_val = binom.test(i, 80, p=1/3)$p.value
  if(p_val < 0.05) {
    cat("At least", i, "correct responses needed (", 
        round(i/80*100, 1), "%)\n")
    break
  }
}

  cat("\nBusiness Recommendation:\n")

min_dim = min(nrow(wine_age)-1, ncol(wine_age)-1)

    ifelse(cramers_v > 0.5, "Large effect (strong association)",
    ifelse(cramers_v > 0.3, "Medium effect (moderate association)",
    ifelse(cramers_v > 0.1, "Small effect (weak association)", 
           "Negligible effect"))), "\n")

# e) Standardized residuals
cat("\n=== e) STANDARDIZED RESIDUALS ===\n")
std_res = (wine_age - ex6_chisq$expected) / sqrt(ex6_chisq$expected)

significant_cells = which(abs(std_res) > 2, arr.ind=TRUE)

    row = significant_cells[i,1]
    col = significant_cells[i,2]
    cat(rownames(wine_age)[row], "-", colnames(wine_age)[col], ": ",
        round(std_res[row,col], 2),
        ifelse(std_res[row,col] > 0, " (more than expected)\n", 
               " (less than expected)\n"))
  }
}

cat("
+---------------------------+----------------+---------------+-----------+---------------------------+
| Situation                 | Data Type      | Same Panel?   | Normal?   | Test                      |
+---------------------------+----------------+---------------+-----------+---------------------------+
| 3+ Independent samples    | Continuous     | No            | Yes       | One-Way ANOVA             |
| 3+ Independent samples    | Continuous     | No            | No        | Kruskal-Wallis            |
| 3+ Dependent samples      | Continuous     | Yes           | Yes       | Repeated Measures ANOVA   |
| 3+ Dependent samples      | Continuous     | Yes           | No        | Friedman Test             |
| Triangle discrimination   | Binary         | -             | -         | Binomial (p=1/3)          |
| Multiple categories       | Categorical    | -             | -         | Chi-square                |
| After ANOVA               | -              | -             | -         | Tukey HSD, Bonferroni     |
| After Kruskal-Wallis      | -              | -             | -         | Dunn's Test               |
| After Friedman            | -              | -             | -         | Nemenyi Test              |
+---------------------------+----------------+---------------+-----------+---------------------------+

KEY DECISION POINTS:

1. How many samples? → 3 or more
2. Same or different panelists? → Determines independence
3. Data type? → Continuous, ordinal, or categorical
4. Normal distribution? → Determines parametric vs non-parametric
5. Significant result? → Proceed to post-hoc tests

EFFECT SIZES:

- Eta-squared (η²): Small = 0.01, Medium = 0.06, Large = 0.14
- Cramér's V: Small = 0.1, Medium = 0.3, Large = 0.5
- Kendall's W: 0-0.3 Weak, 0.3-0.7 Moderate, 0.7-1.0 Strong

POST-HOC TEST SELECTION:

Parametric (after ANOVA):
- Tukey HSD: Most common, controls family-wise error
- Bonferroni: More conservative
- Scheffé: Most conservative, for complex comparisons
- LSD (Fisher): Least conservative, use with caution

Non-parametric:
- Dunn's Test: After Kruskal-Wallis
- Nemenyi Test: After Friedman
- Always use with p-value adjustment (Bonferroni, Holm, etc.)
