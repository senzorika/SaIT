# PART 1: THEORETICAL BACKGROUND
# ============================================================================

# CONSUMER TESTING:
# - Large number of naive consumers (usually 75–300+, depending on design).
# - Main outputs: overall liking, purchase intent, JAR ratings, CATA responses.
# - Usually central location test (CLT) or home-use test (HUT).
#
# JAR (JUST-ABOUT-RIGHT) SCALES:
# - Hybrid scale combining intensity & hedonic information.
# - Examples: "Sweetness" rated on 3-point or 5-point scale:
#   1 = Too weak, 2 = Just-about-right, 3 = Too strong
# - Used to diagnose why overall liking is high/low.
# - Penalty analysis:
#   * Compare mean liking for "JAR" vs "too low" / "too high".
#   * "Drop in liking" (mean difference) + % of consumers in that group.
#   * Helps prioritize reformulation targets.
#
# CATA (CHECK-ALL-THAT-APPLY):
# - Binary responses (0/1) for list of attributes (e.g. 'fruity', 'bitter').
# - Consumers tick all terms that apply to the product.
# - Outputs:
#   * Term frequencies by product.
#   * Association between products and terms (maps).
#   * Links to liking (e.g., mean liking when term is checked vs not checked).
#
# PREFERENCE MAPPING:
# - Goal: Relate consumer liking patterns to product sensory space.
# - Internal Preference Mapping (IPM):
#   * PCA on consumer liking matrix (Products × Consumers).
#   * Consumers are variables; products are observations.
#   * Explore segments and directions of liking.
# - External Preference Mapping (EPM):
#   * Sensory descriptors (e.g. from trained panel or CATA frequencies)
#     create product space.
#   * Regress consumer liking onto sensory dimensions (e.g. PCA axes).
#   * Visualize drivers of liking.

# Example:
# 3 yoghurt samples (A, B, C)
# 200 consumers, each evaluates 1 sample (between-subject design)
# Measures:
# - Overall liking (9-point hedonic scale)
# - JAR for Sweetness (1=Too low, 2=JAR, 3=Too high)
# - JAR for Thickness (1=Too thin, 2=JAR, 3=Too thick)

# --- Simulate data ----------------------------------------------------------

n_cons <- 200
products <- c("A", "B", "C")
jar_levels <- c("Too low", "Just right", "Too high")

# assign each consumer a product
product_assign <- sample(products, n_cons, replace = TRUE,
                         prob = c(0.33, 0.34, 0.33))

# base liking by product
base_liking <- c(A = 6.8, B = 7.5, C = 6.2)

# JAR distributions by product (probabilities)
jar_sweet_probs <- list(
  A = c(0.20, 0.60, 0.20),
  B = c(0.10, 0.70, 0.20),
  C = c(0.40, 0.40, 0.20)
)

jar_thick_probs <- list(
  A = c(0.15, 0.65, 0.20),
  B = c(0.05, 0.75, 0.20),
  C = c(0.25, 0.55, 0.20)
)

# simulate JAR and liking
jar_data <- data.frame(
  Consumer = 1:n_cons,
  Product = factor(product_assign, levels = products)
)

jar_data$JAR_Sweet <- NA
jar_data$JAR_Thick <- NA
jar_data$Liking <- NA

for (p in products) {
  idx <- which(jar_data$Product == p)
  
  jar_data$JAR_Sweet[idx] <- sample(jar_levels, length(idx), replace = TRUE,
                                    prob = jar_sweet_probs[[p]])
  jar_data$JAR_Thick[idx] <- sample(jar_levels, length(idx), replace = TRUE,
                                    prob = jar_thick_probs[[p]])
  
  # liking influenced by product + penalties for JAR deviations
  liking_base <- rnorm(length(idx), mean = base_liking[p], sd = 0.8)
  # small penalties
  penalty_sweet <- ifelse(jar_data$JAR_Sweet[idx] == "Just right", 0,
                          ifelse(jar_data$JAR_Sweet[idx] == "Too low", -0.8, -0.7))
  penalty_thick <- ifelse(jar_data$JAR_Thick[idx] == "Just right", 0,
                          ifelse(jar_data$JAR_Thick[idx] == "Too thin", -0.6, -0.5))
  
  jar_data$Liking[idx] <- liking_base + penalty_sweet + penalty_thick
}

# truncate liking to 1–9
jar_data$Liking <- pmin(pmax(round(jar_data$Liking, 1), 1), 9)

head(jar_data)

# --- JAR Penalty Analysis Function -----------------------------------------

jar_penalty <- function(data, jar_var, liking_var = "Liking",
                        jar_levels = c("Too low", "Just right", "Too high"),
                        min_n = 10) {
  # data: dataframe with columns jar_var and liking_var
  # returns: penalty table
  
  data[[jar_var]] <- factor(data[[jar_var]], levels = jar_levels)
  
  mean_jar <- data %>%
    group_by(.data[[jar_var]]) %>%
    summarise(
      N = n(),
      MeanLiking = mean(.data[[liking_var]]),
      .groups = "drop"
    )
  
  mean_JR <- mean_jar$MeanLiking[mean_jar[[jar_var]] == "Just right"]
  
  mean_jar %>%
    mutate(
      Drop = MeanLiking - mean_JR,
      Drop = round(Drop, 2),
      MeanLiking = round(MeanLiking, 2),
      Percent = round(100 * N / sum(N), 1)
    ) %>%
    filter(N >= min_n)
}

cat("\n=== JAR Penalty Analysis: Sweetness (all products pooled) ===\n")
penalty_sweet_all <- jar_penalty(jar_data, "JAR_Sweet")
print(penalty_sweet_all)

cat("\n=== JAR Penalty Analysis: Thickness (all products pooled) ===\n")
penalty_thick_all <- jar_penalty(jar_data, "JAR_Thick")
print(penalty_thick_all)


# --- Visualization: JAR Penalty Barplot ------------------------------------

plot_jar_penalty <- function(penalty_table, attribute_name) {
  ggplot(penalty_table, aes(x = !!sym(names(penalty_table)[1]),
                            y = -Drop, fill = !!sym(names(penalty_table)[1]))) +
    geom_col() +
    geom_text(aes(label = paste0("Drop=", Drop, "\n", Percent, "%")),
              vjust = -0.2) +
    labs(title = paste("JAR Penalty –", attribute_name),
         x = attribute_name,
         y = "Liking Drop vs 'Just right'") +
    theme_minimal()
}

print(plot_jar_penalty(penalty_thick_all, "Thickness"))

# Example:
# 4 flavoured waters (P1–P4)
# 150 consumers, within-subject design (each evaluates all 4)
# CATA terms: Fruity, Citrus, Sweet, Bitter, Artificial, Fresh, Watery
# Overall liking (9-pt) also collected.

n_cons_cata <- 150
products_cata <- paste0("P", 1:4)
cata_terms <- c("Fruity", "Citrus", "Sweet", "Bitter",
                "Artificial", "Fresh", "Watery")

# design: each consumer evaluates all 4 products
cata_panel <- expand.grid(
  Consumer = 1:n_cons_cata,
  Product = products_cata
)

# base probabilities of CATA terms by product
# (rows products, cols attributes)
cata_prob_matrix <- matrix(
  c(
    # Fruity Citrus Sweet Bitter Artificial Fresh Watery
      0.70, 0.50, 0.65, 0.10, 0.20, 0.60, 0.30,  # P1
      0.50, 0.70, 0.55, 0.15, 0.25, 0.65, 0.40,  # P2
      0.30, 0.20, 0.40, 0.35, 0.45, 0.30, 0.55,  # P3
      0.20, 0.15, 0.25, 0.40, 0.50, 0.25, 0.60   # P4
  ),
  nrow = length(products_cata), byrow = TRUE,
  dimnames = list(products_cata, cata_terms)
)

# base liking by product
base_liking_cata <- c(P1 = 7.3, P2 = 7.6, P3 = 6.2, P4 = 5.8)

# simulate CATA and liking
for (term in cata_terms) {
  cata_panel[[term]] <- 0
}

cata_panel$Liking <- NA

for (p in products_cata) {
  idx <- which(cata_panel$Product == p)
  
  for (term in cata_terms) {
    prob <- cata_prob_matrix[p, term]
    cata_panel[idx, term] <- rbinom(length(idx), 1, prob)
  }
  
  # liking influenced by some attributes (e.g. Fruity+, Fresh+, Artificial-)
  like_base <- rnorm(length(idx), mean = base_liking_cata[p], sd = 0.6)
  
  util <- 0.4 * cata_panel[idx, "Fruity"] +
          0.3 * cata_panel[idx, "Fresh"] -
          0.5 * cata_panel[idx, "Artificial"] -
          0.3 * cata_panel[idx, "Watery"]
  
  cata_panel$Liking[idx] <- like_base + util
}

# truncate liking
cata_panel$Liking <- pmin(pmax(round(cata_panel$Liking, 1), 1), 9)

head(cata_panel)

# a) Regression model using all CATA terms

summary(model_epm_full)

# b) Coefficients
coef_table <- summary(model_epm_full)$coefficients
coef_terms <- coef_table[-1, , drop = FALSE]  # exclude intercept

coef_df <- data.frame(
  Term = rownames(coef_terms),
  Estimate = coef_terms[, "Estimate"],
  p_value = coef_terms[, "Pr(>|t|)"]
)

coef_df <- coef_df %>%
  arrange(desc(Estimate))

cat("\nRegression coefficients for CATA terms (drivers of liking):\n")
print(coef_df)

# c) Compare with Exercise 4 (driver_table$Diff)
drivers_compare <- driver_table %>%
  select(Term, Diff) %>%
  left_join(coef_df, by = "Term")

cat("\nComparison of drivers (Diff vs Regression Estimate):\n")

# d) Final summary table
final_summary <- sensory_data %>%
  mutate(Product = rownames(sensory_data)) %>%
  pivot_longer(cols = all_of(cata_terms), names_to = "Term", values_to = "Freq") %>%
  group_by(Term) %>%
  summarise(
    MaxFreq = max(Freq),
    ProductMax = Product[which.max(Freq)],
    .groups = "drop"
  ) %>%
  left_join(coef_df, by = "Term") %>%
  left_join(driver_table %>% select(Term, Diff), by = "Term") %>%
  mutate(

                            "More is better (higher liking)",

  )

  * Frequency tables, heatmaps.

PREFERENCE MAPPING:
- Internal Preference Mapping (IPM):
  * PCA on liking matrix (products × consumers).
  * Visualizes patterns of preference and consumer segments.
- External Preference Mapping (EPM):
  * Product coordinates from sensory data (e.g., CATA frequencies, trained panel).
  * Regress liking onto sensory space to identify drivers of liking.

KEY DECISION STEPS:
1. What is the goal? Diagnose product issues (JAR), understand sensory space
   (CATA), or relate liking to sensory drivers (PrefMap)?
2. Are we working at consumer level (individual) or product level (means)?
3. For preference mapping:
   - IPM: explores consumer heterogeneity.
   - EPM: connects sensory descriptors to liking.

INTERPRETATION TIPS:
- Large negative JAR penalties with high % of consumers indicate urgent
  reformulation targets (e.g., 'Sweetness too low' with big penalty).
- CATA drivers of liking:
  * Positive drivers: attributes whose presence increases mean liking.
  * Negative drivers: attributes whose presence decreases mean liking.
- Use PrefMap plots to tell a coherent story:
